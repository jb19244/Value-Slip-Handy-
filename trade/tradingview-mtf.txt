// ───────── Typen ─────────
type Zone
    box bx
    float top
    float bot
    int dir

type Liq
    line ln
    label lb
    float px
    int dir

type Mark
    line ln
    label lb

// ───────── Berechnung auf der jeweiligen Zeitebene (nur abgeschlossene Kerzen) ─────────
f_data(int len, float minAtr) =>
    float atr = ta.atr(14)
    float ph = ta.pivothigh(high, len, len)
    float pl = ta.pivotlow(low, len, len)
    var float lastH = na
    var int lastHT = na
    var bool hOpen = false
    var float lastL = na
    var int lastLT = na
    var bool lOpen = false
    var int trend = 0
    bool newPH = not na(ph[1])
    bool newPL = not na(pl[1])
    bool eqh = false
    bool eql = false
    if newPH
        eqh := not na(lastH) and math.abs(ph[1] - lastH) <= atr[1] * 0.1
        lastH := ph[1]
        lastHT := time[len + 1]
        hOpen := true
    if newPL
        eql := not na(lastL) and math.abs(pl[1] - lastL) <= atr[1] * 0.1
        lastL := pl[1]
        lastLT := time[len + 1]
        lOpen := true
    float c = close[1]
    bool bosUp = hOpen and c > lastH
    bool bosDn = lOpen and c < lastL and not bosUp
    bool isCh = false
    float bLvl = na
    int bT = na
    bool obBull = false
    bool obBear = false
    float obTop = na
    float obBot = na
    int obT = na
    if bosUp
        isCh := trend == -1
        trend := 1
        hOpen := false
        bLvl := lastH
        bT := lastHT
        for i = 1 to 30
            if close[i] < open[i]
                obBull := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    if bosDn
        isCh := trend == 1
        trend := -1
        lOpen := false
        bLvl := lastL
        bT := lastLT
        for i = 1 to 30
            if close[i] > open[i]
                obBear := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    bool bullFvg = low[1] > high[3] and (low[1] - high[3]) >= atr[1] * minAtr
    bool bearFvg = high[1] < low[3] and (low[3] - high[1]) >= atr[1] * minAtr
    float fTop = bullFvg ? low[1] : bearFvg ? low[3] : na
    float fBot = bullFvg ? high[3] : bearFvg ? high[1] : na
    int fTime = time[2]
    [bullFvg, bearFvg, fTop, fBot, fTime, newPH, lastH, lastHT, eqh, newPL, lastL, lastLT, eql, bosUp, bosDn, isCh, bLvl, bT, obBull, obBear, obTop, obBot, obT, trend]

f_name(string tf) =>
    int s = timeframe.in_seconds(tf)
    s >= 604800 ? "W" : s >= 86400 ? "D" : s >= 3600 ? str.tostring(math.round(s / 3600.0)) + "h" : str.tostring(math.round(s / 60.0)) + "m"

// ───────── Zeichnen und Pflegen im Chart ─────────
f_clean(Zone[] zs) =>
    if array.size(zs) > 0
        for i = array.size(zs) - 1 to 0
            Zone z = array.get(zs, i)
            bool dead = z.dir > 0 ? close < z.bot : close > z.top
            if dead
                box.delete(z.bx)
                array.remove(zs, i)
    0

f_tf(Zone[] zF, Zone[] zO, Liq[] lq, Mark[] mk, string nm, color tc, bool on, bool nb, bool bullFvg, bool bearFvg, float fTop, float fBot, int fTime, bool newPH, float phV, int phT, bool eqh, bool newPL, float plV, int plT, bool eql, bool bosUp, bool bosDn, bool isCh, float bLvl, int bT, bool obBull, bool obBear, float obTop, float obBot, int obT) =>
    if on
        if nb
            if showFvg and (bullFvg or bearFvg) and not na(fTop) and not na(fTime)
                int d = bullFvg ? 1 : -1
                color fc = d > 0 ? bullCol : bearCol
                box bxF = box.new(fTime, fTop, time, fBot, xloc=xloc.bar_time, extend=extend.right, bgcolor=color.new(fc, fvgTr), border_color=color.new(tc, 45), border_style=line.style_dotted, text=nm + " FVG", text_color=color.new(tc, 10), text_size=size.tiny, text_halign=text.align_left, text_valign=text.align_center)
                array.push(zF, Zone.new(bxF, fTop, fBot, d))
                if array.size(zF) > maxZ
                    Zone oldF = array.shift(zF)
                    box.delete(oldF.bx)
            if showOb and (obBull or obBear) and not na(obTop) and not na(obT)
                int d2 = obBull ? 1 : -1
                color oc = d2 > 0 ? bullCol : bearCol
                box bxO = box.new(obT, obTop, time, obBot, xloc=xloc.bar_time, extend=extend.right, bgcolor=color.new(oc, obTr), border_color=color.new(tc, 0), border_style=line.style_solid, text=nm + " OB", text_color=color.new(tc, 0), text_size=size.tiny, text_halign=text.align_left, text_valign=text.align_center)
                array.push(zO, Zone.new(bxO, obTop, obBot, d2))
                if array.size(zO) > maxZ
                    Zone oldO = array.shift(zO)
                    box.delete(oldO.bx)
            if showLiq and newPH and not na(phV) and not na(phT)
                line lnH = line.new(phT, phV, time, phV, xloc=xloc.bar_time, extend=extend.right, color=color.new(tc, 25), style=line.style_dashed, width=eqh ? 2 : 1)
                label lbH = label.new(time, phV, nm + (eqh ? " BSL =" : " BSL"), xloc=xloc.bar_time, style=label.style_none, textcolor=color.new(tc, 0), size=size.tiny)
                array.push(lq, Liq.new(lnH, lbH, phV, 1))
            if showLiq and newPL and not na(plV) and not na(plT)
                line lnL = line.new(plT, plV, time, plV, xloc=xloc.bar_time, extend=extend.right, color=color.new(tc, 25), style=line.style_dashed, width=eql ? 2 : 1)
                label lbL = label.new(time, plV, nm + (eql ? " SSL =" : " SSL"), xloc=xloc.bar_time, style=label.style_none, textcolor=color.new(tc, 0), size=size.tiny)
                array.push(lq, Liq.new(lnL, lbL, plV, -1))
            while array.size(lq) > maxL
                Liq oldL = array.shift(lq)
                line.delete(oldL.ln)
                label.delete(oldL.lb)
            if showStr and (bosUp or bosDn) and not na(bLvl) and not na(bT)
                color sc = bosUp ? bullCol : bearCol
                line lnS = line.new(bT, bLvl, time, bLvl, xloc=xloc.bar_time, color=sc, style=isCh ? line.style_solid : line.style_dashed, width=isCh ? 2 : 1)
                label lbS = label.new(int(math.avg(bT, time)), bLvl, nm + (isCh ? " CHoCH" : " BOS"), xloc=xloc.bar_time, style=bosUp ? label.style_label_down : label.style_label_up, color=color.new(sc, 100), textcolor=sc, size=size.tiny)
                array.push(mk, Mark.new(lnS, lbS))
                if array.size(mk) > maxM
                    Mark oldM = array.shift(mk)
                    line.delete(oldM.ln)
                    label.delete(oldM.lb)
        // Zonen, durch die der Kurs geschlossen hat, verschwinden
        f_clean(zF)
        f_clean(zO)
        // Liquiditaet: Beschriftung am rechten Rand mitfuehren, abgeholte Level markieren und beenden
        if array.size(lq) > 0
            for i = array.size(lq) - 1 to 0
                Liq q = array.get(lq, i)
                bool swept = q.dir > 0 ? high > q.px : low < q.px
                if swept
                    line.set_extend(q.ln, extend.none)
                    line.set_x2(q.ln, time)
                    line.set_style(q.ln, line.style_dotted)
                    label.set_x(q.lb, time)
                    label.set_text(q.lb, nm + " ✕")
                    array.remove(lq, i)
                else
                    label.set_x(q.lb, time)
    0

f_cnt(Zone[] zs) =>
    int u = 0
    int d = 0
    if array.size(zs) > 0
        for i = 0 to array.size(zs) - 1
            Zone z = array.get(zs, i)
            if z.dir > 0
                u += 1
            else
                d += 1
    str.tostring(u) + "↑ " + str.tostring(d) + "↓"

bool nb2 = ta.change(time(tf2)) != 0
bool nb3 = ta.change(time(tf3)) != 0

// ───────── Übersicht ─────────
f_trendTxt(int t) =>
    t > 0 ? "↑ bullisch" : t < 0 ? "↓ bärisch" : "–"
f_trendCol(int t) =>
    t > 0 ? bullCol : t < 0 ? bearCol : color.gray

f_row(table tbl, int r, string nm, color tc, bool ok, int tr, Zone[] zF, Zone[] zO) =>
    table.cell(tbl, 0, r, nm, text_color=tc, text_size=size.small, text_halign=text.align_left)
    table.cell(tbl, 1, r, ok ? f_trendTxt(tr) : "kleiner als Chart", text_color=ok ? f_trendCol(tr) : color.gray, text_size=size.small)
    table.cell(tbl, 2, r, ok ? f_cnt(zF) : "", text_color=color.silver, text_size=size.small)
    table.cell(tbl, 3, r, ok ? f_cnt(zO) : "", text_color=color.silver, text_size=size.small)

var table tb = table.new(position.top_right, 4, 6, bgcolor=color.new(#101114, 10), border_color=color.new(color.gray, 80), border_width=1, frame_color=color.new(color.gray, 70), frame_width=1)
if showTab and barstate.islast
    table.cell(tb, 0, 0, "Zeit", text_color=color.gray, text_size=size.small, text_halign=text.align_left)
    table.cell(tb, 1, 0, "Struktur", text_color=color.gray, text_size=size.small)
    table.cell(tb, 2, 0, "FVG", text_color=color.gray, text_size=size.small)
    table.cell(tb, 3, 0, "OB", text_color=color.gray, text_size=size.small)
    if on1
        f_row(tb, 1, nm1, c1, ok1, x1, zF1, zO1)
    if on2
        f_row(tb, 2, nm2, c2, ok2, x2, zF2, zO2)
    if on3
        f_row(tb, 3, nm3, c3, ok3, x3, zF3, zO3)
    if on4
        f_row(tb, 4, nm4, c4, ok4, x4, zF4, zO4)
    int up = (ok1 and x1 > 0 ? 1 : 0) + (ok2 and x2 > 0 ? 1 : 0) + (ok3 and x3 > 0 ? 1 : 0) + (ok4 and x4 > 0 ? 1 : 0)
    int dn = (ok1 and x1 < 0 ? 1 : 0) + (ok2 and x2 < 0 ? 1 : 0) + (ok3 and x3 < 0 ? 1 : 0) + (ok4 and x4 < 0 ? 1 : 0)
    int act = (ok1 ? 1 : 0) + (ok2 ? 1 : 0) + (ok3 ? 1 : 0) + (ok4 ? 1 : 0)
    string bias = act > 0 and up == act ? "Alle bullisch: nur Long" : act > 0 and dn == act ? "Alle bärisch: nur Short" : "Gemischt: " + str.tostring(up) + "↑ / " + str.tostring(dn) + "↓"
    color bcol = act > 0 and up == act ? bullCol : act > 0 and dn == act ? bearCol : color.orange
    table.cell(tb, 0, 5, bias, text_color=bcol, text_size=size.small, text_halign=text.align_left)
    table.merge_cells(tb, 0, 5, 3, 5)
