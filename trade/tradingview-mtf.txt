// TRADE CHECK · ZONEN
// Order Blocks, Fair Value Gaps, Liquidität und Struktur (BOS / CHoCH) von bis zu vier Zeitebenen in einem Chart.
// Aufgeräumt: Zonen stehen als schmale Leiste rechts vom Kurs, nur was in Kursnähe liegt wird gezeigt,
// höhere Zeitebenen sind kräftiger als niedrige. Übereinander liegende Zonen werden dunkler: das ist Konfluenz.
// Alles wird nur aus abgeschlossenen Kerzen der jeweiligen Zeitebene berechnet (kein Nachzeichnen).
// Zeitebenen, die kleiner sind als der Chart, werden automatisch ausgelassen.

// ───────── Typen ─────────
type Zone
    box bx
    float top
    float bot
    int dir
    int t0
    color bg
    color bd
    color tx

type Liq
    line ln
    label lb
    float px
    int dir
    color col

type Mark
    line ln
    label lb

// ───────── Lage im Chart ─────────
bool compact = zoneStyle == "Kompakt rechts vom Kurs"
int barMs = timeframe.in_seconds() * 1000
int lEdge = time + barMs
int rEdge = time + barMs * zoneBars
float chartAtr = ta.atr(14)
float nearDist = nearAtr > 0 ? chartAtr * nearAtr : 0.0
color fgCol = chart.fg_color

// ───────── Berechnung auf der jeweiligen Zeitebene (nur abgeschlossene Kerzen) ─────────
f_data(int len, float minAtr) =>
    float atr = ta.atr(14)
    float ph = ta.pivothigh(high, len, len)
    float pl = ta.pivotlow(low, len, len)
    float c = close[1]
    float h1v = high[1]
    float l1v = low[1]
    float h3v = high[3]
    float l3v = low[3]
    float atr1 = atr[1]
    float ph1 = ph[1]
    float pl1 = pl[1]
    int tPiv = time[len + 1]
    int tMid = time[2]
    var float lastH = na
    var int lastHT = na
    var bool hOpen = false
    var float lastL = na
    var int lastLT = na
    var bool lOpen = false
    var int trend = 0
    bool newPH = not na(ph1)
    bool newPL = not na(pl1)
    bool eqh = false
    bool eql = false
    if newPH
        eqh := not na(lastH) and math.abs(ph1 - lastH) <= atr1 * 0.1
        lastH := ph1
        lastHT := tPiv
        hOpen := true
    if newPL
        eql := not na(lastL) and math.abs(pl1 - lastL) <= atr1 * 0.1
        lastL := pl1
        lastLT := tPiv
        lOpen := true
    bool bosUp = hOpen and not na(c) and not na(lastH) and c > lastH
    bool bosDn = lOpen and not na(c) and not na(lastL) and c < lastL and not bosUp
    bool isCh = false
    float bLvl = na
    int bT = na
    bool obBull = false
    bool obBear = false
    float obTop = na
    float obBot = na
    int obT = na
    if bosUp
        isCh := trend == -1
        trend := 1
        hOpen := false
        bLvl := lastH
        bT := lastHT
        for i = 1 to 30
            float ci = close[i]
            float oi = open[i]
            if not na(ci) and ci < oi
                obBull := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    if bosDn
        isCh := trend == 1
        trend := -1
        lOpen := false
        bLvl := lastL
        bT := lastLT
        for i = 1 to 30
            float ci = close[i]
            float oi = open[i]
            if not na(ci) and ci > oi
                obBear := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    bool bullFvg = not na(h3v) and not na(atr1) and l1v > h3v and (l1v - h3v) >= atr1 * minAtr
    bool bearFvg = not na(l3v) and not na(atr1) and h1v < l3v and (l3v - h1v) >= atr1 * minAtr
    float fTop = bullFvg ? l1v : bearFvg ? l3v : na
    float fBot = bullFvg ? h3v : bearFvg ? h1v : na
    int fTime = tMid
    [bullFvg, bearFvg, fTop, fBot, fTime, newPH, lastH, lastHT, eqh, newPL, lastL, lastLT, eql, bosUp, bosDn, isCh, bLvl, bT, obBull, obBear, obTop, obBot, obT, trend]

f_name(string tf) =>
    int s = timeframe.in_seconds(tf)
    s >= 604800 ? "W" : s >= 86400 ? "D" : s >= 3600 ? str.tostring(math.round(s / 3600.0)) + "h" : str.tostring(math.round(s / 60.0)) + "m"

// ───────── Zonen pflegen: rechts mitführen, weit entfernte ausblenden, durchbrochene löschen ─────────
f_upd(array<Zone> zs) =>
    if array.size(zs) > 0
        for i = array.size(zs) - 1 to 0
            Zone z = array.get(zs, i)
            bool dead = z.dir > 0 ? close < z.bot : close > z.top
            if dead
                box.delete(z.bx)
                array.remove(zs, i)
            else
                float dist = close > z.top ? close - z.top : close < z.bot ? z.bot - close : 0.0
                bool far = nearDist > 0 and dist > nearDist
                if compact
                    box.set_left(z.bx, lEdge)
                box.set_right(z.bx, rEdge)
                box.set_bgcolor(z.bx, far ? color(na) : z.bg)
                box.set_border_color(z.bx, far ? color(na) : z.bd)
                box.set_text_color(z.bx, far ? color(na) : z.tx)
    0

f_tf(array<Zone> zF, array<Zone> zO, array<Liq> lq, array<Mark> mk, string nm, int rk, bool enabled, bool nb, bool bullFvg, bool bearFvg, float fTop, float fBot, int fTime, bool newPH, float phV, int phT, bool eqh, bool newPL, float plV, int plT, bool eql, bool bosUp, bool bosDn, bool isCh, float bLvl, int bT, bool obBull, bool obBear, float obTop, float obBot, int obT) =>
    if enabled
        string hal = rk == 1 ? text.align_left : rk == 2 ? text.align_center : text.align_right
        color txc = color.new(fgCol, 40)
        if nb
            if showFvg and (bullFvg or bearFvg) and not na(fTop) and not na(fTime)
                int d = bullFvg ? 1 : -1
                color fc = d > 0 ? bullCol : bearCol
                color bgF = color.new(fc, 93 - rk * 3)
                color bdF = color.new(fc, 100)
                box bxF = box.new(compact ? lEdge : fTime, fTop, rEdge, fBot, xloc=xloc.bar_time, bgcolor=bgF, border_color=bdF, border_width=1, text=nm + " FVG", text_color=txc, text_size=size.tiny, text_halign=hal, text_valign=text.align_center)
                array.push(zF, Zone.new(bxF, fTop, fBot, d, fTime, bgF, bdF, txc))
                if array.size(zF) > maxZ
                    Zone oldF = array.shift(zF)
                    box.delete(oldF.bx)
            if showOb and (obBull or obBear) and not na(obTop) and not na(obT)
                int d2 = obBull ? 1 : -1
                color oc = d2 > 0 ? bullCol : bearCol
                color bgO = color.new(oc, 90 - rk * 4)
                color bdO = color.new(oc, 60 - rk * 8)
                box bxO = box.new(compact ? lEdge : obT, obTop, rEdge, obBot, xloc=xloc.bar_time, bgcolor=bgO, border_color=bdO, border_width=1, text=nm + " OB", text_color=txc, text_size=size.tiny, text_halign=hal, text_valign=text.align_bottom)
                array.push(zO, Zone.new(bxO, obTop, obBot, d2, obT, bgO, bdO, txc))
                if array.size(zO) > maxZ
                    Zone oldO = array.shift(zO)
                    box.delete(oldO.bx)
            color lc = color.new(fgCol, 62 - rk * 8)
            string lst = rk >= 3 ? line.style_solid : line.style_dashed
            if showLiq and newPH and not na(phV) and not na(phT)
                line lnH = line.new(compact ? lEdge : phT, phV, rEdge, phV, xloc=xloc.bar_time, color=lc, style=lst, width=eqh ? 2 : 1)
                label lbH = label.new(rEdge, phV, nm + (eqh ? " EQH" : " Liq ↑"), xloc=xloc.bar_time, style=label.style_label_left, color=color.new(fgCol, 100), textcolor=lc, size=size.tiny)
                array.push(lq, Liq.new(lnH, lbH, phV, 1, lc))
            if showLiq and newPL and not na(plV) and not na(plT)
                line lnL = line.new(compact ? lEdge : plT, plV, rEdge, plV, xloc=xloc.bar_time, color=lc, style=lst, width=eql ? 2 : 1)
                label lbL = label.new(rEdge, plV, nm + (eql ? " EQL" : " Liq ↓"), xloc=xloc.bar_time, style=label.style_label_left, color=color.new(fgCol, 100), textcolor=lc, size=size.tiny)
                array.push(lq, Liq.new(lnL, lbL, plV, -1, lc))
            while array.size(lq) > maxL
                Liq oldL = array.shift(lq)
                line.delete(oldL.ln)
                label.delete(oldL.lb)
            if showStr and (bosUp or bosDn) and not na(bLvl) and not na(bT)
                color sc = bosUp ? bullCol : bearCol
                line lnS = line.new(bT, bLvl, time, bLvl, xloc=xloc.bar_time, color=color.new(sc, 25), style=isCh ? line.style_solid : line.style_dotted, width=1)
                label lbS = label.new(int(math.avg(bT, time)), bLvl, nm + (isCh ? " CHoCH" : " BOS"), xloc=xloc.bar_time, style=bosUp ? label.style_label_down : label.style_label_up, color=color.new(sc, 100), textcolor=sc, size=size.tiny)
                array.push(mk, Mark.new(lnS, lbS))
                if array.size(mk) > maxM
                    Mark oldM = array.shift(mk)
                    line.delete(oldM.ln)
                    label.delete(oldM.lb)
        f_upd(zF)
        f_upd(zO)
        // Liquidität: rechts mitführen, weit entfernte ausblenden, abgeholte Level entfernen
        if array.size(lq) > 0
            for i = array.size(lq) - 1 to 0
                Liq q = array.get(lq, i)
                bool swept = q.dir > 0 ? high > q.px : low < q.px
                if swept
                    line.delete(q.ln)
                    label.delete(q.lb)
                    array.remove(lq, i)
                else
                    bool farL = nearDist > 0 and math.abs(close - q.px) > nearDist
                    if compact
                        line.set_x1(q.ln, lEdge)
                    line.set_x2(q.ln, rEdge)
                    label.set_x(q.lb, rEdge)
                    line.set_color(q.ln, farL ? color(na) : q.col)
                    label.set_textcolor(q.lb, farL ? color(na) : q.col)
    0

f_cnt(array<Zone> zs) =>
    int u = 0
    int d = 0
    if array.size(zs) > 0
        for i = 0 to array.size(zs) - 1
            Zone z = array.get(zs, i)
            if z.dir > 0
                u += 1
            else
                d += 1
    str.tostring(u) + "↑" + str.tostring(d) + "↓"

    int tp = t[1]
    not na(tp) and t != tp

// ───────── Übersicht ─────────
color pTxt = #E8EAED
color pMut = #8A8F98
color pGold = #F5C451
f_trendTxt(int t) =>
    t > 0 ? "▲ bullisch" : t < 0 ? "▼ bärisch" : "–"
f_trendCol(int t) =>
    t > 0 ? bullCol : t < 0 ? bearCol : pMut

f_row(table tbl, int r, string nm, bool ok, int tr, array<Zone> zF, array<Zone> zO) =>
    table.cell(tbl, 0, r, nm, text_color=pTxt, text_size=size.small, text_halign=text.align_left)
    table.cell(tbl, 1, r, ok ? f_trendTxt(tr) : "kleiner als Chart", text_color=ok ? f_trendCol(tr) : pMut, text_size=size.small, text_halign=text.align_left)
    table.cell(tbl, 2, r, ok ? f_cnt(zF) + " · " + f_cnt(zO) : "", text_color=pMut, text_size=size.small, text_halign=text.align_right)

tpos = tabPos == "Oben rechts" ? position.top_right : tabPos == "Unten rechts" ? position.bottom_right : tabPos == "Oben links" ? position.top_left : position.bottom_left
var table tb = table.new(tpos, 3, 6, bgcolor=color.new(#0F1012, 6), border_color=color.new(#2A2D33, 100), border_width=1, frame_color=color.new(#2A2D33, 0), frame_width=1)
if showTab and barstate.islast
    table.cell(tb, 0, 0, "ZONEN", text_color=pGold, text_size=size.small, text_halign=text.align_left)
    table.cell(tb, 1, 0, "Struktur", text_color=pMut, text_size=size.small, text_halign=text.align_left)
    table.cell(tb, 2, 0, "FVG · OB", text_color=pMut, text_size=size.small, text_halign=text.align_right)
    if on1
        f_row(tb, 1, nm1, ok1, x1, zF1, zO1)
    if on2
        f_row(tb, 2, nm2, ok2, x2, zF2, zO2)
    if on3
        f_row(tb, 3, nm3, ok3, x3, zF3, zO3)
    if on4
        f_row(tb, 4, nm4, ok4, x4, zF4, zO4)
    int up = (ok1 and x1 > 0 ? 1 : 0) + (ok2 and x2 > 0 ? 1 : 0) + (ok3 and x3 > 0 ? 1 : 0) + (ok4 and x4 > 0 ? 1 : 0)
    int dn = (ok1 and x1 < 0 ? 1 : 0) + (ok2 and x2 < 0 ? 1 : 0) + (ok3 and x3 < 0 ? 1 : 0) + (ok4 and x4 < 0 ? 1 : 0)
    int act = (ok1 ? 1 : 0) + (ok2 ? 1 : 0) + (ok3 ? 1 : 0) + (ok4 ? 1 : 0)
    string bias = act > 0 and up == act ? "Alle bullisch → nur Long" : act > 0 and dn == act ? "Alle bärisch → nur Short" : "Gemischt " + str.tostring(up) + "▲ " + str.tostring(dn) + "▼ → abwarten"
    color bcol = act > 0 and up == act ? bullCol : act > 0 and dn == act ? bearCol : pGold
    table.cell(tb, 0, 5, bias, text_color=bcol, text_size=size.small, text_halign=text.align_left)
    table.merge_cells(tb, 0, 5, 2, 5)
