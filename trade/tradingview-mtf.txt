// TRADE CHECK · ZONEN
// Order Blocks, Fair Value Gaps, Liquidität und Struktur (BOS / CHoCH) von bis zu vier Zeitebenen in einem Chart.
// Aufgeräumt: Zonen stehen als schmale Leiste rechts vom Kurs, nur was in Kursnähe liegt wird gezeigt,
// höhere Zeitebenen sind kräftiger als niedrige. Übereinander liegende Zonen werden dunkler: das ist Konfluenz.
// Alles wird nur aus abgeschlossenen Kerzen der jeweiligen Zeitebene berechnet (kein Nachzeichnen).
// Zeitebenen, die kleiner sind als der Chart, werden automatisch ausgelassen.
// Nach dem SMC-Leitfaden ergänzt: gebrochene Order Blocks werden zu Breaker Blocks (Rolle getauscht, gestrichelt),
// abgeholte Liquidität mit Schluss zurück wird als Sweep markiert, und Fair Value Gaps schrumpfen beim Füllen.
// Price Action: 15 Kerzen- und Chartmuster werden markiert, jedes mit seinem Ergebnis aus der Rückrechnung
// (15 Min, DAX, Nasdaq, S&P 500, Dow, Gold, EUR/USD, GBP/USD seit 2013; allein, in Trendrichtung, in Zonen, beides).
// Ehrlich: Kein Muster hat nach Kosten bestanden, auch nicht mit Trend oder Zone. Darum sind die Marken grau,
// und der Tooltip sagt, wie das Muster abgeschnitten hat. Für Entscheidungen: Indikator "Trade Check · Ampel".

// ───────── Berechnung auf der jeweiligen Zeitebene (nur abgeschlossene Kerzen) ─────────
f_data(int len, float minAtr) =>
    float atr = ta.atr(14)
    float ph = ta.pivothigh(high, len, len)
    float pl = ta.pivotlow(low, len, len)
    float c = close[1]
    float h1v = high[1]
    float l1v = low[1]
    float h3v = high[3]
    float l3v = low[3]
    float atr1 = atr[1]
    float ph1 = ph[1]
    float pl1 = pl[1]
    int tPiv = time[len + 1]
    int tMid = time[2]
    var float lastH = na
    var int lastHT = na
    var bool hOpen = false
    var float lastL = na
    var int lastLT = na
    var bool lOpen = false
    var int trend = 0
    bool newPH = not na(ph1)
    bool newPL = not na(pl1)
    bool eqh = false
    bool eql = false
    if newPH
        eqh := not na(lastH) and math.abs(ph1 - lastH) <= atr1 * 0.1
        lastH := ph1
        lastHT := tPiv
        hOpen := true
    if newPL
        eql := not na(lastL) and math.abs(pl1 - lastL) <= atr1 * 0.1
        lastL := pl1
        lastLT := tPiv
        lOpen := true
    bool bosUp = hOpen and not na(c) and not na(lastH) and c > lastH
    bool bosDn = lOpen and not na(c) and not na(lastL) and c < lastL and not bosUp
    bool isCh = false
    float bLvl = na
    int bT = na
    bool obBull = false
    bool obBear = false
    float obTop = na
    float obBot = na
    int obT = na
    if bosUp
        isCh := trend == -1
        trend := 1
        hOpen := false
        bLvl := lastH
        bT := lastHT
        for i = 1 to 30
            float ci = close[i]
            float oi = open[i]
            if not na(ci) and ci < oi
                obBull := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    if bosDn
        isCh := trend == 1
        trend := -1
        lOpen := false
        bLvl := lastL
        bT := lastLT
        for i = 1 to 30
            float ci = close[i]
            float oi = open[i]
            if not na(ci) and ci > oi
                obBear := true
                obTop := high[i]
                obBot := low[i]
                obT := time[i]
                break
    bool bullFvg = not na(h3v) and not na(atr1) and l1v > h3v and (l1v - h3v) >= atr1 * minAtr
    bool bearFvg = not na(l3v) and not na(atr1) and h1v < l3v and (l3v - h1v) >= atr1 * minAtr
    float fTop = bullFvg ? l1v : bearFvg ? l3v : na
    float fBot = bullFvg ? h3v : bearFvg ? h1v : na
    int fTime = tMid
    [bullFvg, bearFvg, fTop, fBot, fTime, newPH, lastH, lastHT, eqh, newPL, lastL, lastLT, eql, bosUp, bosDn, isCh, bLvl, bT, obBull, obBear, obTop, obBot, obT, trend]

f_name(string tf) =>
    int s = timeframe.in_seconds(tf)
    s >= 604800 ? "W" : s >= 86400 ? "D" : s >= 3600 ? str.tostring(math.round(s / 3600.0)) + "h" : str.tostring(math.round(s / 60.0)) + "m"

// ───────── Zonen pflegen: rechts mitführen, weit entfernte ausblenden ─────────
// Order Block mit Schluss durch die Zone: wird einmal zum Breaker mit umgekehrter Rolle, beim zweiten Bruch gelöscht.
// Fair Value Gap: schrumpft auf den noch offenen Teil, gelöscht wenn ganz gefüllt oder mit Schluss durchbrochen.
f_upd(array<Zone> zs) =>
    if array.size(zs) > 0
        for i = array.size(zs) - 1 to 0
            Zone z = array.get(zs, i)
            bool through = z.dir > 0 ? close < z.bot : close > z.top
            bool dead = false
            if through and z.ob and not z.brk and showBrk
                z.brk := true
                z.dir := -z.dir
                color nc = z.dir > 0 ? bullCol : bearCol
                z.bg := color.new(nc, 91)
                z.bd := color.new(nc, 45)
                box.set_text(z.bx, z.nm + " Breaker")
                box.set_border_style(z.bx, line.style_dashed)
            else if through
                dead := true
            else if not z.ob and fvgFill
                if z.dir > 0 and low < z.top
                    z.top := math.max(z.bot, low)
                if z.dir < 0 and high > z.bot
                    z.bot := math.min(z.top, high)
                if z.top - z.bot <= 0
                    dead := true
                else
                    box.set_top(z.bx, z.top)
                    box.set_bottom(z.bx, z.bot)
            if dead
                box.delete(z.bx)
                array.remove(zs, i)
            else
                float dist = close > z.top ? close - z.top : close < z.bot ? z.bot - close : 0.0
                bool far = nearDist > 0 and dist > nearDist
                if compact
                    box.set_left(z.bx, lEdge)
                box.set_right(z.bx, rEdge)
                box.set_bgcolor(z.bx, far ? color(na) : z.bg)
                box.set_border_color(z.bx, far ? color(na) : z.bd)
                box.set_text_color(z.bx, far ? color(na) : z.tx)
    0

f_tf(array<Zone> zF, array<Zone> zO, array<Liq> lq, array<Mark> mk, array<label> sw, string nm, int rk, bool enabled, bool nb, bool bullFvg, bool bearFvg, float fTop, float fBot, int fTime, bool newPH, float phV, int phT, bool eqh, bool newPL, float plV, int plT, bool eql, bool bosUp, bool bosDn, bool isCh, float bLvl, int bT, bool obBull, bool obBear, float obTop, float obBot, int obT) =>
    bool swNew = false
    if enabled
        string hal = rk == 1 ? text.align_left : rk == 2 ? text.align_center : text.align_right
        color txc = color.new(fgCol, 40)
        if nb
            if showFvg and (bullFvg or bearFvg) and not na(fTop) and not na(fTime)
                int d = bullFvg ? 1 : -1
                color fc = d > 0 ? bullCol : bearCol
                color bgF = color.new(fc, 93 - rk * 3)
                color bdF = color.new(fc, 100)
                box bxF = box.new(compact ? lEdge : fTime, fTop, rEdge, fBot, xloc=xloc.bar_time, bgcolor=bgF, border_color=bdF, border_width=1, text=nm + " FVG", text_color=txc, text_size=size.tiny, text_halign=hal, text_valign=text.align_center)
                array.push(zF, Zone.new(bxF, fTop, fBot, d, fTime, bgF, bdF, txc, nm, false, false))
                if array.size(zF) > maxZ
                    Zone oldF = array.shift(zF)
                    box.delete(oldF.bx)
            if showOb and (obBull or obBear) and not na(obTop) and not na(obT)
                int d2 = obBull ? 1 : -1
                color oc = d2 > 0 ? bullCol : bearCol
                color bgO = color.new(oc, 90 - rk * 4)
                color bdO = color.new(oc, 60 - rk * 8)
                box bxO = box.new(compact ? lEdge : obT, obTop, rEdge, obBot, xloc=xloc.bar_time, bgcolor=bgO, border_color=bdO, border_width=1, text=nm + " OB", text_color=txc, text_size=size.tiny, text_halign=hal, text_valign=text.align_bottom)
                array.push(zO, Zone.new(bxO, obTop, obBot, d2, obT, bgO, bdO, txc, nm, true, false))
                if array.size(zO) > maxZ
                    Zone oldO = array.shift(zO)
                    box.delete(oldO.bx)
            color lc = color.new(fgCol, 62 - rk * 8)
            string lst = rk >= 3 ? line.style_solid : line.style_dashed
            if showLiq and newPH and not na(phV) and not na(phT)
                line lnH = line.new(compact ? lEdge : phT, phV, rEdge, phV, xloc=xloc.bar_time, color=lc, style=lst, width=eqh ? 2 : 1)
                label lbH = label.new(rEdge, phV, nm + (eqh ? " EQH" : " Liq ↑"), xloc=xloc.bar_time, style=label.style_label_left, color=color.new(fgCol, 100), textcolor=lc, size=size.tiny)
                array.push(lq, Liq.new(lnH, lbH, phV, 1, lc, nm))
            if showLiq and newPL and not na(plV) and not na(plT)
                line lnL = line.new(compact ? lEdge : plT, plV, rEdge, plV, xloc=xloc.bar_time, color=lc, style=lst, width=eql ? 2 : 1)
                label lbL = label.new(rEdge, plV, nm + (eql ? " EQL" : " Liq ↓"), xloc=xloc.bar_time, style=label.style_label_left, color=color.new(fgCol, 100), textcolor=lc, size=size.tiny)
                array.push(lq, Liq.new(lnL, lbL, plV, -1, lc, nm))
            while array.size(lq) > maxL
                Liq oldL = array.shift(lq)
                line.delete(oldL.ln)
                label.delete(oldL.lb)
            if showStr and (bosUp or bosDn) and not na(bLvl) and not na(bT)
                color sc = bosUp ? bullCol : bearCol
                line lnS = line.new(bT, bLvl, time, bLvl, xloc=xloc.bar_time, color=color.new(sc, 25), style=isCh ? line.style_solid : line.style_dotted, width=1)
                label lbS = label.new(int(math.avg(bT, time)), bLvl, nm + (isCh ? " CHoCH" : " BOS"), xloc=xloc.bar_time, style=bosUp ? label.style_label_down : label.style_label_up, color=color.new(sc, 100), textcolor=sc, size=size.tiny)
                array.push(mk, Mark.new(lnS, lbS))
                if array.size(mk) > maxM
                    Mark oldM = array.shift(mk)
                    line.delete(oldM.ln)
                    label.delete(oldM.lb)
        f_upd(zF)
        f_upd(zO)
        // Liquidität: rechts mitführen, weit entfernte ausblenden, abgeholte Level entfernen
        if array.size(lq) > 0
            for i = array.size(lq) - 1 to 0
                Liq q = array.get(lq, i)
                bool swept = q.dir > 0 ? high > q.px : low < q.px
                if swept
                    // Sweep: Docht durch das Level, Schluss wieder davor (Stops abgeräumt, Gegenbewegung möglich)
                    bool back = q.dir > 0 ? close < q.px : close > q.px
                    if back and showSw
                        color swc = q.dir > 0 ? bearCol : bullCol
                        label lbW = label.new(time, q.dir > 0 ? high : low, q.nm + " Sweep", xloc=xloc.bar_time, style=q.dir > 0 ? label.style_label_down : label.style_label_up, color=color.new(swc, 82), textcolor=swc, size=size.tiny)
                        array.push(sw, lbW)
                        swNew := true
                        if array.size(sw) > maxSw
                            label.delete(array.shift(sw))
                    line.delete(q.ln)
                    label.delete(q.lb)
                    array.remove(lq, i)
                else
                    bool farL = nearDist > 0 and math.abs(close - q.px) > nearDist
                    if compact
                        line.set_x1(q.ln, lEdge)
                    line.set_x2(q.ln, rEdge)
                    label.set_x(q.lb, rEdge)
                    line.set_color(q.ln, farL ? color(na) : q.col)
                    label.set_textcolor(q.lb, farL ? color(na) : q.col)
    swNew

f_cnt(array<Zone> zs) =>
    int u = 0
    int d = 0
    if array.size(zs) > 0
        for i = 0 to array.size(zs) - 1
            Zone z = array.get(zs, i)
            if z.dir > 0
                u += 1
            else
                d += 1
    str.tostring(u) + "↑" + str.tostring(d) + "↓"

// ───────── Price Action: alle gängigen Kerzen- und Chartmuster auf dem Chart-Zeitfenster ─────────
// Gleiche Definitionen wie in der Rückrechnung (15 Min, 7 Märkte seit 2013). Jedes Muster trägt sein Ergebnis:
// grün = bestanden, grau = kein Vorteil, rot = verliert nach Kosten. Trend = 4h-Struktur, Zone = offenes 1h-FVG / Order Block.
f_htf(int L) =>
    var int trend = 0
    var float legLow = na
    var float legHigh = na
    var float lastPH = na
    var float lastPL = na
    ph = ta.pivothigh(high, L, L)
    pl = ta.pivotlow(low, L, L)
    if not na(ph)
        lastPH := ph
    if not na(pl)
        lastPL := pl
    if trend == 1
        if close < legLow
            trend := -1
            legHigh := math.max(legHigh, high)
            legLow := low
        else if high > legHigh
            legHigh := high
            if not na(lastPL) and lastPL > legLow
                legLow := lastPL
    else if trend == -1
        if close > legHigh
            trend := 1
            legLow := math.min(legLow, low)
            legHigh := high
        else if low < legLow
            legLow := low
            if not na(lastPH) and lastPH < legHigh
                legHigh := lastPH
    else if not na(lastPH) and not na(lastPL)
        if close > lastPH
            trend := 1
            legLow := lastPL
            legHigh := high
        else if close < lastPL
            trend := -1
            legHigh := lastPH
            legLow := low
    trend[1]
int paTrend = request.security(syminfo.tickerid, "240", f_htf(3), lookahead=barmerge.lookahead_on)

// Ergebnis: Richtung (1 / -1 / 0) und Extrem für den Stop
f_pat(int id) =>
    int d = 0
    float ext = na
    if id == 0
        float lw = math.min(open, close) - low
        float uw = high - math.max(open, close)
        if pR >= 0.6 * pA
            if lw >= 2 * pB and lw >= 0.6 * pR and close - low >= 0.66 * pR
                d := 1
                ext := low
            else if uw >= 2 * pB and uw >= 0.6 * pR and high - close >= 0.66 * pR
                d := -1
                ext := high
    else if id == 1
        if pB >= 0.5 * pA
            if f_dn(1) and f_up(0) and close >= open[1] and open <= close[1]
                d := 1
                ext := math.min(low, low[1])
            else if f_up(1) and f_dn(0) and close <= open[1] and open >= close[1]
                d := -1
                ext := math.max(high, high[1])
    else if id == 2
        if high[1] < high[2] and low[1] > low[2]
            if close > high[2]
                d := 1
                ext := low[1]
            else if close < low[2]
                d := -1
                ext := high[1]
    else if id == 3
        if high > high[1] and low < low[1] and pR >= pA
            float pos = (close - low) / pR
            if pos > 0.75 and close > high[1]
                d := 1
                ext := low
            else if pos < 0.25 and close < low[1]
                d := -1
                ext := high
    else if id == 4
        if f_bd(2) >= 0.6 * pA and f_bd(1) <= 0.3 * f_bd(2)
            if f_dn(2) and f_up(0) and close > (open[2] + close[2]) / 2
                d := 1
                ext := math.min(low, low[1], low[2])
            else if f_up(2) and f_dn(0) and close < (open[2] + close[2]) / 2
                d := -1
                ext := math.max(high, high[1], high[2])
    else if id == 5
        bool allUp = true
        bool allDn = true
        for k = 0 to 2
            float rk = high[k] - low[k]
            allUp := allUp and f_up(k) and f_bd(k) >= 0.4 * pA and high[k] - close[k] <= 0.25 * rk
            allDn := allDn and f_dn(k) and f_bd(k) >= 0.4 * pA and close[k] - low[k] <= 0.25 * rk
        if allUp and close[1] > close[2] and close > close[1] and open[1] >= open[2] and open >= open[1]
            d := 1
            ext := low[2]
        else if allDn and close[1] < close[2] and close < close[1] and open[1] <= open[2] and open <= open[1]
            d := -1
            ext := high[2]
    else if id == 6
        if f_dn(1) and f_up(0) and math.abs(low[1] - low) <= 0.05 * pA and f_bd(1) >= 0.3 * pA
            d := 1
            ext := math.min(low, low[1])
        else if f_up(1) and f_dn(0) and math.abs(high[1] - high) <= 0.05 * pA and f_bd(1) >= 0.3 * pA
            d := -1
            ext := math.max(high, high[1])
    else if id == 7
        if f_bd(1) >= 0.5 * pA
            if f_dn(1) and f_up(0) and open <= close[1] and close > (open[1] + close[1]) / 2 and close < open[1]
                d := 1
                ext := math.min(low, low[1])
            else if f_up(1) and f_dn(0) and open >= close[1] and close < (open[1] + close[1]) / 2 and close > open[1]
                d := -1
                ext := math.max(high, high[1])
    else if id == 8
        if f_bd(1) >= 0.6 * pA and pB <= 0.5 * f_bd(1)
            if f_dn(1) and f_up(0) and open >= close[1] and close <= open[1]
                d := 1
                ext := low[1]
            else if f_up(1) and f_dn(0) and open <= close[1] and close >= open[1]
                d := -1
                ext := high[1]
    else if id == 9
        float dojiR = high[1] - low[1]
        if dojiR > 0 and f_bd(1) <= 0.1 * dojiR
            if close[2] < close[3] and close[3] < close[4] and close > high[1]
                d := 1
                ext := math.min(low, low[1])
            else if close[2] > close[3] and close[3] > close[4] and close < low[1]
                d := -1
                ext := math.max(high, high[1])
    else if id == 10
        if pR >= 1.2 * pA and pB >= 0.9 * pR
            d := close > open ? 1 : -1
            ext := close > open ? low : high
    else if id == 11
        if high[1] < high[2] and low[1] > low[2]
            if low < low[1] and close > low[1] and close > open
                d := 1
                ext := low
            else if high > high[1] and close < high[1] and close < open
                d := -1
                ext := high
    else if id == 12
        for n = 3 to 8
            float mv = close[n + 1] - open[n + 3]
            float fh = high[1]
            float fl = low[1]
            for k = 1 to n
                fh := math.max(fh, high[k])
                fl := math.min(fl, low[k])
            if mv >= 2 * pA and fl >= open[n + 3] + 0.5 * mv and close > fh
                d := 1
                ext := fl
                break
            if mv <= -2 * pA and fh <= open[n + 3] + 0.5 * mv and close < fl
                d := -1
                ext := fh
                break
    [d, ext]
