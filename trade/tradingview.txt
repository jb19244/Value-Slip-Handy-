f_struct() =>
    ph = ta.pivothigh(high, 2, 2)
    pl = ta.pivotlow(low, 2, 2)
    var float lh = na
    var float ll = na
    var int d = 0
    if not na(ph)
        lh := ph
    if not na(pl)
        ll := pl
    if not na(lh) and close > lh
        d := 1
    else if not na(ll) and close < ll
        d := -1
    d[1]
h4dir = request.security(syminfo.tickerid, "240", f_struct(), lookahead = barmerge.lookahead_on)

f_draw(Sig s, float gLo, float gHi) =>
    col = f_tierColor(s.tier)
    x2 = bar_index + win
    array.push(sgB, box.new(bar_index - 2, gHi, x2, gLo, border_color = color.new(col, 100), bgcolor = color.new(col, 90)))
    array.push(sgL, line.new(bar_index, s.entry, x2, s.entry, color = col, width = 2))
    array.push(sgL, line.new(bar_index, s.stop, x2, s.stop, color = cBad, style = line.style_dashed))
    array.push(sgL, line.new(bar_index, s.tp1, x2, s.tp1, color = color.new(cGood, 30), style = line.style_dotted))
    array.push(sgL, line.new(bar_index, s.tp2, x2, s.tp2, color = cGood, width = 2))
    risk = math.abs(s.entry - s.stop)
    eur = acct * riskP / 100
    qty = risk > 0 ? eur / risk : na
    stat = s.tier == "CRÈME" ? "In Bewährung: 96 Trades, 60 % Gewinner mit Teilverkauf, im Härtetest Ø +0,19 R (1 Std, 2 Jahre)" : s.tier == "TREND" ? "Mit Tagestrend bei VIX < 20: rund 50 % Gewinner, kein Vorteil belegt" : "Ohne Beleg: gegen den Tagestrend"
    txt = (s.tier == "CRÈME" ? "★ " : "") + s.tier + (s.dir > 0 ? " LONG" : " SHORT") + "\nEinstieg " + str.tostring(s.entry, format.mintick) + " · Stop " + str.tostring(s.stop, format.mintick) + "\nTP1 " + str.tostring(s.tp1, format.mintick) + " · Ziel " + str.tostring(s.tp2, format.mintick) + " (CRV " + str.tostring(s.rr, "#.0") + ")"
    array.push(sgT, label.new(bar_index, s.stop, txt, style = s.dir > 0 ? label.style_label_up : label.style_label_down, color = color.new(col, 10), textcolor = color.black, size = size.small, tooltip = s.name + "\n" + stat + "\nRisiko " + str.tostring(eur, "#") + " € = " + str.tostring(qty, "#.##") + " Einheiten Basiswert\nPlan: bei TP1 die Hälfte verkaufen und den Stop auf Einstand ziehen, Rest bis zum Ziel, spätestens 21:45 schließen."))
    // ältere Signale aus dem Chart nehmen, damit er lesbar bleibt
    while array.size(sgT) > maxSig
        label.delete(array.shift(sgT))
        box.delete(array.shift(sgB))
        line.delete(array.shift(sgL))
        line.delete(array.shift(sgL))
        line.delete(array.shift(sgL))
        line.delete(array.shift(sgL))
    true

newSig = false

f_addLevel(bool cond, int d, float lv, string nm) =>
    if cond and not na(lv)
        pierced = d > 0 ? low < lv : high > lv
        back = d > 0 ? close > lv : close < lv
        float c1p = close[1]
        float l1p = low[1]
        float h1p = high[1]
        prevP = d > 0 ? (c1p < lv and l1p < lv) : (c1p > lv and h1p > lv)
        if (pierced and back) or (prevP and back)
            ext = d > 0 ? math.min(low, l1p) : math.max(high, h1p)
            ref = d > 0 ? refHi5 : refLo5
            array.push(pend, Pend.new(bar_index, d, lv, ext, ref, 0, dayHiBefore, dayLoBefore, nm, bDay))
            true
        else
            false
    else
        false

if true
    // offene Kandidaten weiterführen
    if array.size(pend) > 0
        for k = array.size(pend) - 1 to 0
            s = array.get(pend, k)
            drop = bar_index - s.i > win or s.day != bDay
            if not drop and s.broke == 0
                if (s.dir > 0 and low < s.ext - atr * 0.05) or (s.dir < 0 and high > s.ext + atr * 0.05)
                    s.ext := s.dir > 0 ? math.min(s.ext, low) : math.max(s.ext, high)
                    if math.abs(close - s.level) > atr * 0.8 and (close - s.level) * s.dir < 0
                        drop := true
            if not drop and s.broke == 0 and (close - s.ref) * s.dir > 0
                s.broke := bar_index
            if not drop and s.broke > 0
                // größte FVG der Verdrängung seit dem Sweep
                float gLo = na
                float gHi = na
                minG = atr * 0.3
                // Rueckgriff in Kerzen: von der juengsten (0) bis zur aeltesten erlaubten Kerze. Nie negativ, auch wenn Sweep und Bruch auf derselben Kerze liegen.
                int oMax = bar_index - math.max(s.i + 2, s.broke - 3)
                if oMax >= 0
                    for o = 0 to oMax
                        float cl = na
                        float ch = na
                        float h2 = high[o + 2]
                        float l2 = low[o + 2]
                        float h0 = high[o]
                        float l0 = low[o]
                        if s.dir > 0 and l0 > h2
                            cl := h2
                            ch := l0
                        if s.dir < 0 and h0 < l2
                            cl := h0
                            ch := l2
                        if not na(cl) and ch - cl >= minG and (na(gLo) or ch - cl > gHi - gLo)
                            gLo := cl
                            gHi := ch
                if not na(gLo)
                    entry = (gLo + gHi) / 2
                    stop = s.ext - s.dir * atr * 0.1
                    risk = (entry - stop) * s.dir
                    if risk > atr * 0.3 and risk < atr * 4
                        // Ziel: nächste Liquidität mit CRV 2–6 (Asien-Extrem, Vortag, Tagesextrem vor dem Sweep)
                        float best = na
                        float bestRR = na
                        c1 = s.dir > 0 ? asHi : asLo
                        c2 = s.dir > 0 ? pdh : pdl
                        c3 = s.dir > 0 ? s.dHi : s.dLo
                        for cv in array.from(c1, c2, c3)
                            if not na(cv)
                                rrv = (cv - entry) * s.dir / risk
                                if rrv >= 2 and rrv <= 6 and (na(bestRR) or rrv < bestRR)
                                    best := cv
                                    bestRR := rrv
                        tp2 = na(best) ? entry + s.dir * risk * 2.5 : best
                        rr = (tp2 - entry) * s.dir / risk
                        withB = dBias == s.dir
                        tier = withB and vixHi ? "CRÈME" : withB ? "TREND" : "GEGEN TREND"
                        ok = showSweep and (tier == "CRÈME" or (not onlyCreme and (tier == "TREND" or showCounter)))
                        if ok
                            sg = Sig.new(bar_index, s.dir, entry, stop, entry + s.dir * risk, tp2, rr, tier, "Sweep " + s.name + " → FVG", not na(best))
                            f_draw(sg, gLo, gHi)
                            last := sg
                            newSig := true
                            alert((tier == "CRÈME" ? "★ CRÈME " : tier + " ") + syminfo.ticker + (s.dir > 0 ? " LONG" : " SHORT") + " · Limit " + str.tostring(entry, format.mintick) + " · SL " + str.tostring(stop, format.mintick) + " · TP1 " + str.tostring(entry + s.dir * risk, format.mintick) + " · TP2 " + str.tostring(tp2, format.mintick), alert.freq_once_per_bar_close)
                    drop := true
                else if bar_index - s.broke >= 3
                    drop := true
            if drop
                array.remove(pend, k)
    // neue Sweeps nur in der Killzone an Wochentagen
    if inKZ
        if bMin >= 480 and not swAL
            swAL := f_addLevel(true, 1, asLo, "Asien-Tief")
        if bMin >= 480 and not swAH
            swAH := f_addLevel(true, -1, asHi, "Asien-Hoch")
        if not swPL
            swPL := f_addLevel(true, 1, pdl, "Vortagestief")
        if not swPH
            swPH := f_addLevel(true, -1, pdh, "Vortageshoch")
        if bMin >= 870 and not swLL
            swLL := f_addLevel(true, 1, lnLo, "London-Tief")
        if bMin >= 870 and not swLH
            swLH := f_addLevel(true, -1, lnHi, "London-Hoch")
        if not swWL
            swWL := f_addLevel(true, 1, pwl, "Vorwochentief")
        if not swWH
            swWH := f_addLevel(true, -1, pwh, "Vorwochenhoch")

        f_row(10, "IBS (Tag)", str.tostring(ibsNow, "#.00") + (ibsSig ? (ibsNow < 0.1 ? " → Kauf vor Schluss · 66 %" : " → leichtes Signal · 63 %") : " · Signal unter 0,20"), ibsSig ? cGood : cMuted)
