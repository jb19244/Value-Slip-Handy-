type EqSide
    bool         isHigh
    array<float> pivP
    array<int>   pivI
    array<line>  lns
    array<label> lbs
    array<float> lvs
    array<int>   xs

eqNew(bool isHigh) =>
    EqSide.new(isHigh, array.new<float>(), array.new<int>(), array.new<line>(), array.new<label>(), array.new<float>(), array.new<int>())

// neues bestaetigtes Pivot pruefen: gleiches Hoch/Tief im Lookback, ohne hoeheres Hoch (tieferes Tief) dazwischen
method addPivot(EqSide s, float pv, bool show, color col) =>
    if show and not na(pv)
        int   idx = bar_index - eqPiv
        float tol = nz(atr14[eqPiv]) * eqTol
        while s.pivI.size() > 0 and idx - s.pivI.first() > eqLen
            s.pivI.shift()
            s.pivP.shift()
        int n = s.pivI.size()
        if n > 0
            for j = n - 1 to 0
                float p   = s.pivP.get(j)
                int   pi  = s.pivI.get(j)
                float ext = s.isHigh ? math.max(p, pv) : math.min(p, pv)
                bool  blocked = false
                int   kEnd = bar_index - pi - 1
                if kEnd >= eqPiv + 1
                    for k = eqPiv + 1 to kEnd
                        if s.isHigh ? high[k] > ext + tol : low[k] < ext - tol
                            blocked := true
                            break
                if blocked
                    break
                if math.abs(p - pv) <= tol
                    s.lns.push(line.new(pi, p, idx, pv, color = col, width = eqW, style = eqSty))
                    s.lbs.push(eqLbl ? label.new(math.round((pi + idx) / 2), ext, s.isHigh ? "EQH" : "EQL", style = s.isHigh ? label.style_label_down : label.style_label_up, color = color(na), textcolor = col, size = size.small) : label(na))
                    s.lvs.push(ext)
                    s.xs.push(idx)
                    break
        s.pivP.push(pv)
        s.pivI.push(idx)
    s

// geholt oder aelter als der Lookback: entfernen
method sweep(EqSide s) =>
    int n = s.lns.size()
    if n > 0
        for j = n - 1 to 0
            float lv    = s.lvs.get(j)
            bool  swept = eqDel and bar_index > s.xs.get(j) + eqPiv and (s.isHigh ? high > lv : low < lv)
            bool  old   = bar_index - s.xs.get(j) > eqLen
            if swept or old
                s.lns.remove(j).delete()
                label lb = s.lbs.remove(j)
                if not na(lb)
                    lb.delete()
                s.lvs.remove(j)
                s.xs.remove(j)
    while s.lns.size() > EQ_MAX
        s.lns.shift().delete()
        label lb0 = s.lbs.shift()
        if not na(lb0)
            lb0.delete()
        s.lvs.shift()
        s.xs.shift()
    s

var EqSide eqH = eqNew(true)
var EqSide eqL = eqNew(false)
eqH.addPivot(ph, eqhOn, eqhCol)
eqL.addPivot(pl, eqlOn, eqlCol)
eqH.sweep()
eqL.sweep()

type Lvl
    line  ln = na
    label lb = na
    float px = na

method run(Lvl this, bool ch, bool show, string tf, color col, string txt) =>
    if show and timeframe.in_seconds() < timeframe.in_seconds(tf)
        if ch
            if not na(this.ln)
                this.ln.delete()
            if not na(this.lb)
                this.lb.delete()
            this.px := open
            this.ln := line.new(bar_index, open, bar_index + 1, open, color = col, width = oLw, style = oSt, extend = extend.right)
            this.lb := label.new(bar_index + 3, open, txt, style = label.style_label_left, color = color(na), textcolor = col, size = size.small)
        if barstate.islast and not na(this.lb)
            this.lb.set_x(bar_index + 3)
            this.lb.set_text(oPx ? txt + "  " + str.tostring(this.px, format.mintick) : txt)
    this

type Sess
    box        bx  = na
    array<box> all = na
    float      hi  = na
    float      lo  = na

method run(Sess s, bool show, string sess, color col, string name) =>
    bool inS = not na(time(timeframe.period, sess, TZ))
    color bc = color.new(col, 30)
    if show and timeframe.isintraday
        if inS and not inS[1]
            s.hi := high
            s.lo := low
            s.bx := box.new(bar_index, high, bar_index, low, border_color = bc, border_width = 1, bgcolor = col, text = sTxt ? name : "", text_color = bc, text_size = size.small, text_halign = text.align_left, text_valign = text.align_top)
            s.all.push(s.bx)
            if s.all.size() > sDays
                s.all.shift().delete()
        else if inS and not na(s.bx)
            s.hi := math.max(s.hi, high)
            s.lo := math.min(s.lo, low)
            s.bx.set_top(s.hi)
            s.bx.set_bottom(s.lo)
            s.bx.set_right(bar_index)
    s

var Sess seA = Sess.new(all = array.new<box>())

trendOf() =>
    float f = ta.ema(close, tFast)
    float s = ta.ema(close, tSlow)
    close > s and f > s ? 1 : close < s and f < s ? -1 : 0

trChart = trendOf()
tr1 = request.security(syminfo.tickerid, tf1, trendOf())

tfName(string tf) =>
    tf == "D" or tf == "1D" ? "1 Tag" : tf == "W" or tf == "1W" ? "1 Woche" : tf == "M" or tf == "1M" ? "1 Monat" : str.endswith(tf, "S") ? tf : str.endswith(tf, "D") ? str.replace(tf, "D", " Tage") : str.endswith(tf, "W") ? str.replace(tf, "W", " Wochen") : str.endswith(tf, "M") ? str.replace(tf, "M", " Monate") : str.tonumber(tf) % 60 == 0 ? str.tostring(str.tonumber(tf) / 60) + " Std" : tf + " Min"

        cDir := trChart
    else
        cnt  := 0
        cDir := 0

    cLbs.push(label.new(bar_index, low, str.tostring(cnt), xloc = xloc.bar_index, yloc = yloc.belowbar, style = label.style_none, textcolor = cDir == 1 ? tUp : tDn, size = cSiz))

        cLbs.shift().delete()
