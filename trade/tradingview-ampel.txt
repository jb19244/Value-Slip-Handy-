// TRADE CHECK · AMPEL
// Eine Antwort statt vieler Zonen: Gibt es JETZT einen Trade, der sich nachweislich lohnt?
//
// Warum keine Zonen mehr (FVG, Order Blocks, Liquidität über 15 Min / 1 h / 4 h)?
// Wir haben genau die Frage "korrigiert der Kurs weiter oder dreht er in meine Richtung?" mit
// 5-Minuten-Kursen seit 2013 auf DAX, Nasdaq, S&P 500, Dow, Gold, EUR/USD und GBP/USD nachgerechnet:
// 4h-Richtung + Rücklauf in die günstige Hälfte + 15-Min-Wende (CHoCH) + Ziel 4h-Hoch.
// Ergebnis: auf keinem der 7 Märkte ein Plus nach Spread, nicht besser als auf Zufallskursen.
// Die Zonen sehen überzeugend aus, sagen aber die nächste Bewegung nicht voraus.
//
// Die Ampel zeigt darum nur Regeln, die die Rückrechnung bestanden haben, und rechnet sie auf
// deinem Chart selbst nach:
//  · IBS (Tag): Index schließt im unteren Fünftel der Tagesspanne über dem 200-Tage-Schnitt → kaufen
//  · Turnaround Tuesday (US-Indizes): schwacher Montag → kaufen
//  · Drei Verlusttage im Aufwärtstrend → kaufen
//  · Liquiditäts-Sweep (SMC): Tag fällt unter das 30-Tage-Tief und schließt wieder darüber, im Aufwärtstrend → kaufen
//    (seit 2000 auf 16 Märkten: 1.201 Trades, 46 % Treffer, Ø +0,21 R; auf 15 Min dagegen im Minus)
//  · DAX-Trend-Ausbruch (9:00–9:15, Order ab 9:35) und DAX-Nachmittag (15:30–16:30, Order ab 16:40)
// Grün heißt: Signal nach einer belegten Regel und auf diesem Markt bisher im Plus. Sonst nicht.
// Alles nur aus abgeschlossenen Kerzen; Tagessignale gelten, wenn der Tag so schließt.
// ═══════════════════════════════════════════════════════════════════════════════════════════════

f_fmt(float v) => str.tostring(v, format.mintick)
f_pc(float v) => (v >= 0 ? "+" : "−") + str.tostring(math.abs(v), "#.##") + " %"
f_r(float v) => (v >= 0 ? "+" : "−") + str.tostring(math.abs(v), "#.##") + " R"
f_q(int w, int n) => n > 0 ? str.tostring(math.round(w * 100.0 / n)) + " %" : "–"

// ═════════ TAGESREGELN (auf Tageskerzen gerechnet, ein Platz je Markt wie im Push-Dienst) ═════════
// kind: 1 = IBS, 2 = Turnaround Tuesday, 3 = Drei Verlusttage. Einstieg zum Schluss des Signaltags.
f_day(bool uI, bool uT, bool uD, bool uS, bool usIdx) =>
    rng = high - low
    ib = rng > 0 ? (close - low) / rng : 0.5
    s200 = ta.sma(close, 200)
    s5 = ta.sma(close, 5)
    a14 = ta.atr(14)
    mon = dayofweek == dayofweek.monday
    sI = uI and not na(s200) and close > s200 and ib < 0.2
    sT = uT and usIdx and mon and close < close[1] and ib < 0.3
    sD = uD and not na(s200) and close > s200 and close < close[1] and close[1] < close[2] and close[2] < close[3]
    lo30 = ta.lowest(low, 30)[1]
    sS = uS and not na(s200) and not na(lo30) and close > s200 and low < lo30 and close > lo30
    var int kind = 0
    var float e = na
    var float st = na
    var int held = 0
    var int eTime = 0
    var float tg = na
    var int n4 = 0
    var float s4 = 0.0
    var int w4 = 0
    var int n1 = 0
    var float s1 = 0.0
    var int w1 = 0
    var int n2 = 0
    var float s2 = 0.0
    var int w2 = 0
    var int n3 = 0
    var float s3 = 0.0
    var int w3 = 0
    float xp = na
    int xk = 0
    if kind != 0
        held += 1
        if low <= st
            xp := math.min(st, open)
        else if kind == 4 and high >= tg
            xp := math.max(tg, open)
        else if kind == 4 and held >= 10
            xp := close
        else if kind == 1 and (ib > 0.8 or held >= 5)
            xp := close
        else if kind == 2 and (close > high[1] or held >= 4)
            xp := close
        else if kind == 3 and (close > s5 or held >= 7)
            xp := close
        if not na(xp)
            pct = (xp / e - 1) * 100
            xk := kind
            if kind == 1
                n1 += 1
                s1 += pct
                w1 += pct > 0 ? 1 : 0
            else if kind == 2
                n2 += 1
                s2 += pct
                w2 += pct > 0 ? 1 : 0
            else if kind == 3
                n3 += 1
                s3 += pct
                w3 += pct > 0 ? 1 : 0
            else
                n4 += 1
                s4 += pct
                w4 += pct > 0 ? 1 : 0
            kind := 0
    int newK = 0
    if kind == 0 and na(xp) and not na(a14)
        newK := sI ? 1 : sT ? 2 : sD ? 3 : sS ? 4 : 0
        if newK != 0
            kind := newK
            e := close
            st := newK == 4 ? math.min(low - 0.2 * a14, close - 0.5 * a14) : close - 3 * a14
            tg := newK == 4 ? close + 2 * (close - st) : na
            held := 0
            eTime := time
    // Wie tief müsste der Tag schließen, damit IBS < 0,2 wird?
    ibsLvl = low + 0.2 * rng
    [ib, close, s200, s5, a14, sI, sT, sD, kind, e, st, held, eTime, newK, xk, ibsLvl, mon, n1, s1, w1, n2, s2, w2, n3, s3, w3, sS, tg, lo30, n4, s4, w4, low]

[dIb, dClose, dS200, dS5, dAtr, dSI, dST, dSD, dKind, dE, dSt, dHeld, dETime, dNewK, dXk, dIbsLvl, dMon, n1, s1, w1, n2, s2, w2, n3, s3, w3, dSS, dTg, dLo30, n4, s4, w4, dLow] = request.security(syminfo.tickerid, "D", f_day(useIbs, useTT, useD3, useSw, isUS), lookahead=barmerge.lookahead_off)
m1 = n1 > 0 ? s1 / n1 : 0.0
m2 = n2 > 0 ? s2 / n2 : 0.0
m3 = n3 > 0 ? s3 / n3 : 0.0
m4 = n4 > 0 ? s4 / n4 : 0.0
kindName(int k) => k == 1 ? "IBS" : k == 2 ? "Turnaround Tuesday" : k == 3 ? "Drei Verlusttage" : k == 4 ? "Liquiditäts-Sweep" : ""
kindExit(int k) => k == 4 ? "Ziel (2 R) und Stop gleich als Order, spätestens nach 10 Tagen verkaufen" : k == 1 ? "raus bei Schluss im oberen Fünftel der Tagesspanne, spätestens nach 5 Tagen" : k == 2 ? "raus bei Schluss über dem Vortageshoch, spätestens nach 4 Tagen" : "raus bei Schluss über dem 5-Tage-Schnitt, spätestens nach 7 Tagen"

okHere(int n, float m) => n < 15 or m > 0

            aO := open
    if aS == 0 and aN > 0 and bm >= 1000 and bm < 1045
        aDir := aO > pC ? 1 : aO < pC ? -1 : 0
        buf = 0.03 * pAtr
        aPx := aDir > 0 ? aHi + buf : aLo - buf
        aStop := aDir > 0 ? aLo - buf : aHi + buf
        rk = math.abs(aPx - aStop)
        if aDir == 0
            aS := 3

        else if (open - aPx) * aDir > 0
            aS := 3

        else if not (rk > 3.6 and rk <= 3 * pAtr)
            aS := 3

    if aS == 1
        if bm >= 1045

        else if aDir > 0 ? high >= aPx : low <= aPx
            aS := 2
            aE := aDir > 0 ? math.max(aPx, open) : math.min(aPx, open)
            f_mark(aDir > 0, aDir > 0 ? low : high, aDir > 0 ? "▲ DAX-Nachmittag" : "▼ DAX-Nachmittag")
    if aS == 2

            xq := open

        if not na(xq)

    lnE = float(na)

    if isDax and intraOk and mS == 2

    else if isDax and intraOk and aS == 2
        cr = (close - aE) * aDir / math.abs(aE - aStop)
        head := "DAX-NACHMITTAG LÄUFT  " + f_r(cr)

        why := (aDir > 0 ? "Long" : "Short") + " seit " + f_fmt(aE) + ". Stop " + f_fmt(aStop) + " bleibt, um 17:25 schließen."
        lnE := aE

    else if dKind != 0
        pcNow = (close / dE - 1) * 100
